/** * MATCO Production Plan & QC — Backend (Google Apps Script) * ------------------------------------------------------------ * ทำหน้าที่เป็น API หลังบ้าน อ่าน/เขียนข้อมูลลง Google Sheet 2 ชีท: * - Items : ข้อมูลแผนผลิต/ผล QC ทั้งหมด (แทนที่ localStorage เดิม) * - AccessLog : บันทึกการเข้าใช้งานทุกครั้ง (login, addPlan, submitActual, approve, reject) * * วิธีติดตั้ง (ทำครั้งเดียว): * 1. สร้าง Google Sheet ใหม่ 1 ไฟล์ (ชื่ออะไรก็ได้ เช่น "MATCO_QC_DATA") * 2. เปิดเมนู Extensions > Apps Script แล้ววางโค้ดไฟล์นี้ทั้งหมดแทนโค้ดเริ่มต้น * 3. กด Deploy > New deployment > เลือกประเภท "Web app" * - Execute as: Me * - Who has access: Anyone (จำเป็น เพราะเรียกจาก GitHub Pages ซึ่งไม่มี Google Login) * 4. คัดลอก Web app URL ที่ได้ ไปวางในตัวแปร API_URL ของไฟล์ index.html * 5. ทุกครั้งที่แก้โค้ดไฟล์นี้ ต้องกด Deploy > Manage deployments > แก้ไข (เวอร์ชันใหม่) ใหม่ทุกครั้ง * ไม่เช่นนั้น URL เดิมจะยังใช้โค้ดเวอร์ชันเก่าอยู่ */ var SHEET_ITEMS = 'Items'; var SHEET_LOG = 'AccessLog'; var ITEM_HEADERS = ['id','model','color','dueDate','note','status','createdBy','createdAt','actualDate','actualNote','fileUrl','qcBy','qcAt']; function doPost(e) { var lock = LockService.getScriptLock(); lock.waitLock(15000); // กันข้อมูลชนกันเวลามีหลายคนกดพร้อมกัน try { var body = JSON.parse(e.postData.contents); var action = body.action; var userName = body.user || 'unknown'; var dept = body.department || ''; var payload = body.payload || {}; logAccess(userName, dept, action, payload); // บันทึก Log ทุก action พร้อมแผนก // ตรวจสิทธิ์เบื้องต้นตามแผนก (ยังอิงจากค่าที่ frontend ส่งมาเอง ไม่ใช่ authentication จริง // แต่กันการเรียก action ผิดแผนกโดยไม่ตั้งใจ / กันการแก้ hash ตรง ๆ ได้ในระดับหนึ่ง) var qcOnlyActions = ['approve', 'reject']; var prodOnlyActions = ['addPlan', 'submitActual']; if (qcOnlyActions.indexOf(action) !== -1 && dept !== 'qc') { return jsonOut({ ok: false, error: 'สิทธิ์ไม่เพียงพอ: การดำเนินการนี้สำหรับแผนก QC เท่านั้น' }); } if (prodOnlyActions.indexOf(action) !== -1 && dept !== 'production') { return jsonOut({ ok: false, error: 'สิทธิ์ไม่เพียงพอ: การดำเนินการนี้สำหรับแผนกผลิตเท่านั้น' }); } var result; switch (action) { case 'login': result = { ok: true }; break; case 'list': result = { ok: true, data: listItems() }; break; case 'addPlan': result = addPlan(payload); break; case 'submitActual': result = submitActual(payload); break; case 'approve': result = setStatus(payload.id, 'approved', userName); break; case 'reject': result = setStatus(payload.id, 'rejected', userName); break; default: result = { ok: false, error: 'unknown action: ' + action }; } return jsonOut(result); } catch (err) { return jsonOut({ ok: false, error: String(err) }); } finally { lock.releaseLock(); } } function doGet(e) { // เผื่อเรียกทดสอบผ่าน URL ตรง ๆ ในเบราว์เซอร์ return jsonOut({ ok: true, data: listItems() }); } function jsonOut(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function getSheet(name) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName(name); if (!sh) { sh = ss.insertSheet(name); if (name === SHEET_ITEMS) sh.appendRow(ITEM_HEADERS); if (name === SHEET_LOG) sh.appendRow(['timestamp', 'user', 'department', 'action', 'detail']); } return sh; } function listItems() { var sh = getSheet(SHEET_ITEMS); var values = sh.getDataRange().getValues(); if (values.length < 2) return []; var headers = values.shift(); return values .filter(function (row) { return row[0] !== ''; }) // ข้ามแถวว่าง .map(function (row) { var obj = {}; headers.forEach(function (h, i) { var v = row[i]; // แปลง Date object ของ Sheet กลับเป็น string ISO เพื่อให้ frontend ใช้งานง่าย obj[h] = (v instanceof Date) ? v.toISOString() : v; }); return obj; }); } function addPlan(payload) { var sh = getSheet(SHEET_ITEMS); var item = payload.item || {}; var row = ITEM_HEADERS.map(function (h) { return item[h] !== undefined && item[h] !== null ? item[h] : ''; }); sh.appendRow(row); return { ok: true }; } function findRowById(sh, id) { var data = sh.getDataRange().getValues(); for (var r = 1; r < data.length; r++) { if (String(data[r][0]) === String(id)) return r + 1; // เลขแถวจริง (1-indexed, +1 เพราะมี header) } return -1; } function submitActual(payload) { var sh = getSheet(SHEET_ITEMS); var row = findRowById(sh, payload.id); if (row === -1) return { ok: false, error: 'ไม่พบรายการ id: ' + payload.id }; setCell(sh, row, 'fileUrl', payload.fileUrl || ''); setCell(sh, row, 'actualDate', payload.actualDate || ''); setCell(sh, row, 'actualNote', payload.actualNote || ''); setCell(sh, row, 'status', 'waiting'); return { ok: true }; } function setStatus(id, status, userName) { var sh = getSheet(SHEET_ITEMS); var row = findRowById(sh, id); if (row === -1) return { ok: false, error: 'ไม่พบรายการ id: ' + id }; setCell(sh, row, 'status', status); setCell(sh, row, 'qcBy', userName); setCell(sh, row, 'qcAt', new Date().toISOString()); return { ok: true }; } function setCell(sh, row, colName, value) { var col = ITEM_HEADERS.indexOf(colName) + 1; sh.getRange(row, col).setValue(value); } function logAccess(userName, dept, action, payload) { var sh = getSheet(SHEET_LOG); var detail = ''; try { detail = payload ? JSON.stringify(payload).slice(0, 500) : ''; } catch (e) {} sh.appendRow([new Date().toISOString(), userName, dept, action, detail]); }
